% 01 — Inputs and PPI initialization
\[
\begin{aligned}
\bm X&=[x_{sg}]\in\mathbb R^{B\times N},
&\bm H_s^{(0)}&=\bm E,\\[-1mm]
\left.\bm E_{g:}\right|_{\mathrm{init}}
&=
\begin{cases}
\bm p_g, & g\ \text{mapped by PPI node2vec},\\
\bm\varepsilon_g, & \text{otherwise},
\end{cases}
&\bm E&\ \text{remains trainable.}
\end{aligned}
\]

\newpage

% 02 — Content projections
\[
\begin{aligned}
\bm Q_s^{(\ell,h)}&=\bm H_s^{(\ell-1)}\bm W_Q^{(\ell,h)},\\
\bm K_s^{(\ell,h)}&=\bm H_s^{(\ell-1)}\bm W_K^{(\ell,h)},\\
\bm V_s^{(\ell,h)}&=\bm H_s^{(\ell-1)}\bm W_V^{(\ell,h)},
\end{aligned}
\qquad
\bm Q_s^{(\ell,h)},\bm K_s^{(\ell,h)},\bm V_s^{(\ell,h)}
\in\mathbb R^{N\times d_h}.
\]

\newpage

% 03 — Full expression-derived TUPE bias
\[
\begin{aligned}
\bm Q_{x,s}^{h}&=\bm x_s\bm W_{Qx}^{h},
&\bm K_{x,s}^{h}&=\bm x_s\bm W_{Kx}^{h},\\[1mm]
\bm B_s^{h}
&=\frac{\bm Q_{x,s}^{h}\bigl(\bm K_{x,s}^{h}\bigr)^{\!\top}}
{\sqrt{2d_h}}
&\in\mathbb R^{N\times N},
\end{aligned}
\qquad
\bm W_{Qx}^{h},\bm W_{Kx}^{h}\in\mathbb R^{1\times d_h}.
\]

\newpage

% 04 — Full patient-conditioned attention
\[
\begin{aligned}
\bm S_s^{(\ell,h)}
&=\frac{\bm Q_s^{(\ell,h)}\bigl(\bm K_s^{(\ell,h)}\bigr)^{\!\top}}
{\sqrt{2d_h}}+\bm B_s^h,\\[1mm]
\bm A_s^{(\ell,h)}
&=\operatorname{softmax}_{\mathrm{row}}\!\left(\bm S_s^{(\ell,h)}\right),\\[1mm]
\bm O_s^{(\ell,h)}
&=\operatorname{Drop}_{\mathrm{attn}}\!\left(\bm A_s^{(\ell,h)}\right)
\bm V_s^{(\ell,h)}.
\end{aligned}
\]

\newpage

% HERO — Complete TUPE signature in one continuous block
\[
\boxed{
\begin{aligned}
\bm Q_{x,s}^{h}&=\bm x_s\bm W_{Qx}^{h},
&\bm K_{x,s}^{h}&=\bm x_s\bm W_{Kx}^{h},\\[-1mm]
\bm B_s^{h}
&=\frac{\bm Q_{x,s}^{h}(\bm K_{x,s}^{h})^{\!\top}}{\sqrt{2d_h}},
&\bm S_s^{(\ell,h)}
&=\frac{\bm Q_s^{(\ell,h)}(\bm K_s^{(\ell,h)})^{\!\top}}{\sqrt{2d_h}}+\bm B_s^h,\\[-1mm]
\bm A_s^{(\ell,h)}
&=\operatorname{softmax}_{\mathrm{row}}\!\left(\bm S_s^{(\ell,h)}\right),
&\bm O_s^{(\ell,h)}
&=\operatorname{Drop}_{\mathrm{attn}}\!\left(\bm A_s^{(\ell,h)}\right)\bm V_s^{(\ell,h)}.
\end{aligned}}
\]

\newpage

% 05 — Multi-head merge
\[
\bm M_s^{(\ell)}
=\operatorname{Concat}_{h=1}^{H}\!\left(\bm O_s^{(\ell,h)}\right)
\bm W_O^{(\ell)}
\in\mathbb R^{N\times d_e},
\qquad
\bm W_O^{(\ell)}\in\mathbb R^{d_m\times d_e}.
\]

\newpage

% 06 — Exact post-norm encoder update
\[
\begin{aligned}
\bm U_s^{(\ell)}
&=\operatorname{LN}_1^{(\ell)}\!\left[
\bm H_s^{(\ell-1)}+\operatorname{Drop}_1^{(\ell)}\!\left(\bm M_s^{(\ell)}\right)
\right],\\[1mm]
\bm F_s^{(\ell)}
&=\operatorname{Drop}_{ff}^{(\ell)}\!\left[
\operatorname{ReLU}\!\left(\bm U_s^{(\ell)}\bm W_1^{(\ell)}\right)
\right]\bm W_2^{(\ell)},\\[1mm]
\bm H_s^{(\ell)}
&=\operatorname{LN}_2^{(\ell)}\!\left[
\bm U_s^{(\ell)}+\operatorname{Drop}_2^{(\ell)}\!\left(\bm F_s^{(\ell)}\right)
\right].
\end{aligned}
\]

\newpage

% 07 — Final norm and average pooling
\[
\widetilde{\bm H}_s
=\operatorname{LN}_{f}\!\left(\bm H_s^{(L)}\right)
\in\mathbb R^{N\times d_e},
\qquad
\bm z_s
=\frac{1}{N}\sum_{g=1}^{N}\widetilde{\bm H}_{s,g:}
\in\mathbb R^{d_e}.
\]

\newpage

% 08 — Mask-aware two-logit heads
\[
\begin{aligned}
m_{s,1}&=\mathbf 1[c_s\in\{\mathrm{MCI},\mathrm{AD}\}],\\
m_{s,2}&=\mathbf 1[c_s\in\{\mathrm{CTL},\mathrm{AD}\}],\\
m_{s,3}&=\mathbf 1[c_s\in\{\mathrm{CTL},\mathrm{MCI}\}],
\end{aligned}
\qquad
\bm o_{s,t}=f_t(\bm z_s)\in\mathbb R^2
\quad\text{only if }m_{s,t}=1.
\]

\newpage

% 09 — Per-task and weighted total loss
\[
\begin{aligned}
I_{b,t}&=\{s\in b:m_{s,t}=1\},
&\ell_{b,t}&=\frac{1}{|I_{b,t}|}\sum_{s\in I_{b,t}}
\operatorname{CE}(\bm o_{s,t},y_{s,t}),\\[1mm]
\mathcal L_b
&=\frac{\sum_{t\in\mathcal A_b}\lambda_t\ell_{b,t}}
{\sum_{t\in\mathcal A_b}\lambda_t},
&\bm\lambda&=(0.50,0.25,0.25).
\end{aligned}
\]

\newpage

% A — Compact full TUPE signature
\[
\boxed{
\begin{aligned}
\bm B_s^h
&=\frac{\bm Q_{x,s}^{h}\bigl(\bm K_{x,s}^{h}\bigr)^{\!\top}}{\sqrt{2d_h}},\\[-1mm]
\bm O_s^h
&=\operatorname{Drop}\!\left[
\operatorname{softmax}_{\mathrm{row}}\!\left(
\frac{\bm Q_E^h(\bm K_E^h)^{\!\top}}{\sqrt{2d_h}}+\bm B_s^h
\right)\right]\bm V_E^h.
\end{aligned}}
\]

\newpage

% B — Compact end-to-end model
\[
\boxed{
\widetilde{\bm H}_s
=\operatorname{LN}_f\!\left[
\operatorname{TxT}_{\mathrm{postnorm}}(\bm E;\bm B_s)
\right],
\qquad
\bm z_s=\frac1N\sum_g\widetilde{\bm H}_{s,g:},
\qquad
\bm o_{s,t}=f_t(\bm z_s)\ \text{if }m_{s,t}=1.}
\]

\newpage

% C — Compact multitask objective
\[
\boxed{
\ell_t
=\frac1{n_t}\sum_{s:m_{s,t}=1}\operatorname{CE}(\bm o_{s,t},y_{s,t}),
\qquad
\mathcal L
=\frac{\sum_{t\in\mathcal A}\lambda_t\ell_t}
{\sum_{t\in\mathcal A}\lambda_t}.}
\]
